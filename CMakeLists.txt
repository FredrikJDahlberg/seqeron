# The cluster tier's C++ build (doc/future-arch.md §11 step 7a.3). Configures standing alone —
# `cmake -S cluster -B <dir>` — and is added by the root build, which needs everything here for its
# own targets too: this file owns the toolchain both sides share (the Aeron and GoogleTest fetches,
# the SBE tool, the warning/sanitizer flags) because core owns the Aeron dependency (§9.4). It
# declares no simdfix and generates no application codecs.
cmake_minimum_required(VERSION 3.28)
project(phixeron_core VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(FetchContent)
include(CheckCXXCompilerFlag)

check_cxx_compiler_flag("-msse4.1" PHIXERON_HAS_MSSE41)

add_library(phixeron_flags INTERFACE)
target_compile_options(phixeron_flags INTERFACE
    -Wall -Wextra
    $<$<CONFIG:Debug>:-g -O0 -fsanitize=address -fno-omit-frame-pointer>
    $<$<CONFIG:Release>:-O3>
    $<$<BOOL:${PHIXERON_HAS_MSSE41}>:-msse4.1>
)
target_link_options(phixeron_flags INTERFACE
    $<$<CONFIG:Debug>:-fsanitize=address>
)

option(PHIXERON_COVERAGE "Instrument for coverage" OFF)
if(PHIXERON_COVERAGE)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang") # matches both Clang and AppleClang
        target_compile_options(phixeron_flags INTERFACE -fprofile-instr-generate -fcoverage-mapping)
        target_link_options(phixeron_flags INTERFACE -fprofile-instr-generate)
    elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_compile_options(phixeron_flags INTERFACE --coverage)
        target_link_options(phixeron_flags INTERFACE --coverage)
    else()
        message(WARNING "PHIXERON_COVERAGE=ON but ${CMAKE_CXX_COMPILER_ID} has no known coverage flags")
    endif()
endif()

# Aeron — fetched and built from source.
set(AERON_TESTS            OFF CACHE BOOL "" FORCE)
set(AERON_SAMPLES          OFF CACHE BOOL "" FORCE)
set(AERON_BUILD_SAMPLES    OFF CACHE BOOL "" FORCE)
set(AERON_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(BUILD_AERON_ARCHIVE_API ON  CACHE BOOL "" FORCE)

# PATCH_COMMAND corrects Aeron's own declared OUTPUTs for the C archive codecs, which name a
# directory generateCCodecs never writes.
FetchContent_Declare(
    aeron
    GIT_REPOSITORY https://github.com/real-logic/aeron.git
    GIT_TAG        1.51.0
    GIT_SHALLOW    TRUE
    SYSTEM
    UPDATE_COMMAND ""
    PATCH_COMMAND  ${CMAKE_COMMAND} -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/FixAeronCodecOutputs.cmake
)
FetchContent_MakeAvailable(aeron)

# Google Test — simdfix (fetched by the root build) uses find_package(GTest); we use FetchContent so
# both work: FetchContent registers targets that find_package then resolves.
FetchContent_Declare(
    googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG        v1.16.0
    GIT_SHALLOW    TRUE
)
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)

# SBE tool — downloaded once at configure time
find_package(Java REQUIRED COMPONENTS Runtime)

set(SBE_VERSION    "1.38.1")
set(SBE_JAR        "${CMAKE_BINARY_DIR}/sbe-all-${SBE_VERSION}.jar")
# Each schema generates into its own namespace directory below, and each codegen command wipes that
# directory before running: SBE only ever writes, so without this a message deleted from a schema leaves
# its header behind and stays compilable. The stamp file is the command's only tracked output, so CMake
# cannot notice a stale header on its own.
#
# Two include roots, not one, and that is the module boundary rather than tidiness: `core` holds the
# schemas the cluster tier owns (frame, replay, cluster) and `app` the three application payloads
# (session, basicdata, order), generated by the root build. phixeron_core includes only the first, so a
# cluster-tier file that reached for an application codec fails to build here rather than at the repo
# split — the same guarantee the separate source include roots already give for hand-written headers.
set(SBE_CORE_DIR "${CMAKE_BINARY_DIR}/generated/sbe/core")

if(NOT EXISTS "${SBE_JAR}")
    message(STATUS "Downloading SBE tool ${SBE_VERSION}…")
    file(DOWNLOAD
        "https://repo1.maven.org/maven2/uk/co/real-logic/sbe-all/${SBE_VERSION}/sbe-all-${SBE_VERSION}.jar"
        "${SBE_JAR}"
        TLS_VERIFY ON
        SHOW_PROGRESS
    )
endif()

# The root build generates the application schemas with the same tool.
if(NOT PROJECT_IS_TOP_LEVEL)
    set(SBE_JAR "${SBE_JAR}" PARENT_SCOPE)
endif()

# SBE C++ codegen for sbe-cluster.xml
set(CLUSTER_SBE_SCHEMA "${CMAKE_CURRENT_SOURCE_DIR}/src/main/sbe/sbe-cluster.xml")
set(CLUSTER_SBE_STAMP  "${SBE_CORE_DIR}/.cluster_sbe_stamp")

add_custom_command(
    OUTPUT  ${CLUSTER_SBE_STAMP}
    COMMAND ${CMAKE_COMMAND} -E rm -rf ${SBE_CORE_DIR}/org_limitless_phixeron_cluster_sbe
    COMMAND ${CMAKE_COMMAND} -E make_directory ${SBE_CORE_DIR}
    COMMAND ${Java_JAVA_EXECUTABLE}
                -Dsbe.output.dir=${SBE_CORE_DIR}
                -Dsbe.target.language=Cpp
                -Dsbe.cpp.namespaces.collapse=false
                -jar ${SBE_JAR}
                ${CLUSTER_SBE_SCHEMA}
    COMMAND ${CMAKE_COMMAND} -E touch ${CLUSTER_SBE_STAMP}
    DEPENDS ${CLUSTER_SBE_SCHEMA} ${SBE_JAR}
    COMMENT "Generating SBE C++ codecs from sbe-cluster.xml"
    VERBATIM
)

add_custom_target(GenerateClusterSbeCodecs DEPENDS ${CLUSTER_SBE_STAMP})

# SBE C++ codegen for sbe-replay.xml (the node-local replay control protocol)
set(REPLAY_SBE_SCHEMA "${CMAKE_CURRENT_SOURCE_DIR}/src/main/sbe/sbe-replay.xml")
set(REPLAY_SBE_STAMP  "${SBE_CORE_DIR}/.replay_sbe_stamp")

add_custom_command(
    OUTPUT  ${REPLAY_SBE_STAMP}
    COMMAND ${CMAKE_COMMAND} -E rm -rf ${SBE_CORE_DIR}/org_limitless_phixeron_sbe_replay
    COMMAND ${CMAKE_COMMAND} -E make_directory ${SBE_CORE_DIR}
    COMMAND ${Java_JAVA_EXECUTABLE}
                -Dsbe.output.dir=${SBE_CORE_DIR}
                -Dsbe.target.language=Cpp
                -Dsbe.cpp.namespaces.collapse=false

                ${REPLAY_SBE_SCHEMA}

add_custom_target(GenerateReplaySbeCodecs DEPENDS ${REPLAY_SBE_STAMP})

# SBE C++ codegen for sbe-frame.xml (the Unsequenced/Sequenced envelope pair)
set(FRAME_SBE_SCHEMA "${CMAKE_CURRENT_SOURCE_DIR}/src/main/sbe/sbe-frame.xml")
set(FRAME_SBE_STAMP  "${SBE_CORE_DIR}/.frame_sbe_stamp")

add_custom_command(

# phixeron_core — header-only interface library for the cluster tier (doc/future-arch.md §11 step 1).
# Its include root is cluster/src/main/cpp, so nothing linking only this can reach a product header,
# and it carries no simdfix and no generated FIX codecs: its SBE root is SBE_CORE_DIR, which holds
# only frame/replay/cluster, so the cluster tier decodes no FIX by construction rather than by habit.
add_library(phixeron_core INTERFACE)
target_include_directories(phixeron_core INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src/main/cpp>
    $<INSTALL_INTERFACE:include>
)
# SYSTEM: SBE-tool output. Regenerating from the schema is the only way to change these, so their
# diagnostics are noise — -Wtype-limits fires on every get*AsJsonEscapedString wherever char is
# unsigned (aarch64). Keeps -Wall -Wextra live on the hand-written sources.
target_include_directories(phixeron_core SYSTEM INTERFACE
    $<BUILD_INTERFACE:${SBE_CORE_DIR}>
)
target_link_libraries(phixeron_core INTERFACE
    phixeron_flags
    aeron_client_wrapper
)

# core_tests — the cluster tier's own suite. Links phixeron_core only, so a core file that reached
# for a product header would fail to build here rather than at extraction.
enable_testing()

add_executable(core_tests
    src/test/cpp/org/limitless/phixeron/sequencer/ClusterCodecTest.cpp
    src/test/cpp/org/limitless/phixeron/sequencer/ConformanceTest.cpp
    src/test/cpp/org/limitless/phixeron/sequencer/ClusterStreamSenderTest.cpp
    src/test/cpp/org/limitless/phixeron/sequencer/FrameStartPositionTest.cpp
    src/test/cpp/org/limitless/phixeron/sequencer/PortLayoutTest.cpp
    src/test/cpp/org/limitless/phixeron/replayer/client/ReplayerRecoveryTest.cpp
    src/test/cpp/org/limitless/phixeron/replayer/client/RecoveryProgressPolicyTest.cpp
    src/test/cpp/org/limitless/phixeron/fix/GatewayRecoveryStallPolicyTest.cpp
    src/main/cpp/org/limitless/phixeron/sequencer/ClusterStreamSender.hpp
    src/main/cpp/org/limitless/phixeron/fix/GatewayRecoveryStallPolicy.hpp
    src/main/cpp/org/limitless/phixeron/util/Logger.hpp
)
target_link_libraries(core_tests PRIVATE
    phixeron_core
    aeron_archive_wrapper
    aeron_archive_c_client
    aeron
    GTest::gtest_main
)
add_dependencies(core_tests GenerateReplaySbeCodecs GenerateClusterSbeCodecs
    GenerateFrameSbeCodecs)

check_cxx_compiler_flag("-Wcharacter-conversion" PHIXERON_HAS_WCHARACTER_CONVERSION)
if(PHIXERON_HAS_WCHARACTER_CONVERSION)
    target_compile_options(core_tests PRIVATE -Wno-character-conversion)
endif()

include(GoogleTest)
gtest_discover_tests(core_tests)
